cmake_minimum_required(VERSION 3.24)
project(MinimaDock VERSION 0.2.29 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dock_model src/taskbar/state_machine.cpp)
target_include_directories(dock_model PUBLIC src)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(model_tests tests/state_machines/model_tests.cpp)
target_link_libraries(model_tests PRIVATE dock_model)
add_test(NAME model_tests COMMAND model_tests)
add_executable(calendar_tests tests/state_machines/calendar_tests.cpp)
target_include_directories(calendar_tests PRIVATE src)
add_test(NAME calendar_model COMMAND calendar_tests)
add_executable(media_marquee_tests tests/state_machines/media_marquee_tests.cpp)
target_include_directories(media_marquee_tests PRIVATE src)
add_test(NAME media_marquee COMMAND media_marquee_tests)
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_executable(MinimaDock WIN32 src/app/main.cpp src/rendering/preview.cpp src/rendering/calendar.cpp src/windows/media.cpp src/windows/inventory.cpp)
  target_link_libraries(MinimaDock PRIVATE dock_model d2d1 dwrite dwmapi shell32 ole32 advapi32 windowscodecs uuid runtimeobject)
  add_executable(MinimaDockGuard src/recovery/guard.cpp)
  target_link_libraries(MinimaDockGuard PRIVATE advapi32 shell32)
  add_executable(MinimaDockTrial WIN32 src/recovery/taskbar_trial.cpp)
  target_link_libraries(MinimaDockTrial PRIVATE advapi32 shell32 dwmapi)
  add_executable(work_area_tests tests/integration/work_area_tests.cpp)
  target_include_directories(work_area_tests PRIVATE src)
  target_link_libraries(work_area_tests PRIVATE dwmapi)
  add_test(NAME work_area_geometry COMMAND work_area_tests)
  add_executable(work_area_probe tests/integration/work_area_probe.cpp)
  target_include_directories(work_area_probe PRIVATE src)
  target_link_libraries(work_area_probe PRIVATE dwmapi)
  add_executable(pin_state_tests tests/integration/pin_state_tests.cpp src/rendering/calendar.cpp src/windows/media.cpp src/windows/inventory.cpp)
  target_link_libraries(pin_state_tests PRIVATE dock_model d2d1 dwrite dwmapi shell32 ole32 advapi32 windowscodecs uuid gdi32 runtimeobject)
  add_test(NAME pinned_window_layout COMMAND pin_state_tests)
  add_executable(media_ui_tests tests/integration/media_ui_tests.cpp src/rendering/calendar.cpp src/windows/media.cpp src/windows/inventory.cpp)
  target_link_libraries(media_ui_tests PRIVATE dock_model d2d1 dwrite dwmapi shell32 ole32 advapi32 windowscodecs uuid gdi32 runtimeobject)
  add_test(NAME media_native_ui COMMAND media_ui_tests)
  add_executable(media_tests tests/integration/media_tests.cpp src/windows/media.cpp)
  target_include_directories(media_tests PRIVATE src)
  target_link_libraries(media_tests PRIVATE ole32 shell32 runtimeobject uuid)
  add_executable(MinimaDockGuardTest src/recovery/guard.cpp)
  target_compile_definitions(MinimaDockGuardTest PRIVATE MINIMA_GUARD_TEST)
  target_link_libraries(MinimaDockGuardTest PRIVATE advapi32 shell32)
  add_executable(MinimaDockGuardFixture tests/integration/guard_fixture.cpp)
  add_executable(guard_tests tests/integration/guard_tests.cpp)
  add_test(NAME guard_scenarios COMMAND guard_tests)
  set_tests_properties(guard_scenarios PROPERTIES TIMEOUT 120)
  if(MINGW)
    target_link_options(MinimaDock PRIVATE -municode -static)
    target_link_options(MinimaDockGuard PRIVATE -municode -static)
    target_link_options(MinimaDockTrial PRIVATE -municode -static)
    target_link_options(pin_state_tests PRIVATE -static)
    target_link_options(media_tests PRIVATE -static)
    target_link_options(media_ui_tests PRIVATE -static)
    target_link_options(MinimaDockGuardTest PRIVATE -municode -static)
    target_link_options(MinimaDockGuardFixture PRIVATE -municode -static)
    target_link_options(guard_tests PRIVATE -municode -static)
  endif()
  if(MSVC)
    set_property(TARGET MinimaDock MinimaDockGuard MinimaDockTrial model_tests dock_model PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(MinimaDock PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app/app.manifest")
  endif()
  install(TARGETS MinimaDock MinimaDockGuard MinimaDockTrial RUNTIME DESTINATION .)
  install(DIRECTORY tools/recovery/ DESTINATION recovery)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
  install(DIRECTORY third_party/licenses/ DESTINATION third_party/licenses)
endif()
